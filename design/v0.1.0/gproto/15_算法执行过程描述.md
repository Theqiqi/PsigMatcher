## ParsePatternBytes

tokens = split(normalize(raw))        # 去空格/逗号；或按 2 字符切分连续串
for t in tokens:
    (b,m) = TokenToByteMask(t)

## FindAll
```
if mask.empty() or size < mask.size(): return {}
m = mask.size()
for i in 0..(size-m):
    ok = true
    for j in 0..(m-1):
        if (data[i+j] & mask[j]) != (bytes[j] & mask[j]): ok=false; break
    if ok: offsets.push(i)
    if limit reached: break
return offsets
```

## ScanProcesses（接口层编排）
```
cancel_flag = 全局原子标志
results = []
for pid in req.pids:
    if cancel_flag: break
    h = OpenForRead(pid); if !h: continue
    name,path = ProcessNamePath(pid)
    modules = EnumModules(pid)
    regions = EnumRegions(h)
    for r in regions:
        if cancel_flag: break
        buf = ReadMemory(h, r.base, r.size); if empty: continue
        for p in req.patterns (valid):
            if cancel_flag: break
            offs = FindAll(buf, p.bytes, p.mask, remainingLimit)
            for o in offs:
                mod = LocateModule(r.base+o, modules)
                results.push(MatchResult{...,address=r.base+o, matchedBytes=hex(buf+o, len), hitTime=now})
                hits++
            progress({pid, done, total, hits})
    CloseHandle(h)
return results
```
